  155| class PartitionRefinement(Generic[T]):
  156|     """Maintain and refine a partition of a set of items into subsets.
  157|     Space usage for a partition of n items is O(n), and each refine operation
  158|     takes time proportional to the size of its argument.
  159| 
  160|     Adapted from code by D. Eppstein:
  161|     https://www.ics.uci.edu/~eppstein/PADS/PartitionRefinement.py
  162|     """
  163| 
  164|     __slots__: Tuple[str, ...] = ("_sets", "_partition")
  165| 
  166|     _sets: Dict[int, Set[T]]
  167|     _partition: Dict[T, int]
  168| 
  169|     def __init__(self, items: Iterable[T]) -> None:
  170|         """Create a new partition refinement data structure for the given
  171|         items. Initially, all items belong to the same subset.
  172|         """
  173|         S = set(items)
  174|         self._sets = {id(S): S}
  175|         self._partition = {x: id(S) for x in S}
  176| 
  177|     def get_set_by_id(self, id: int) -> Set[T]:
  178|         """Return the set in the partition corresponding to id."""
  179|         return self._sets[id]
  180| 
  181|     def get_set_ids(self) -> Iterable[int]:
  182|         """Return set ids corresponding to the internal partition."""
  183|         return self._sets.keys()
  184| 
  185|     def get_sets(self) -> Iterable[Set[T]]:
  186|         """Return sets corresponding to the internal partition."""
  187|         return self._sets.values()
