  578|         if self.allow_partial:
  579|             # In the case of a partial DFA, we want to try to condense
  580|             # possible trap states before the main minify operation.
  581|             graph = self._get_digraph()
  582|             live_states = get_reachable_nodes(graph, [self.initial_state])
  583|             non_trap_states = get_reachable_nodes(
  584|                 graph, self.final_states, reversed=True
  585|             )
  586| 
  587|             reachable_states = live_states & non_trap_states
  588|             reachable_states.add(self.initial_state)
  589|         else:
  590|             # Compute reachable states and final states
  591|             bfs_states = self.__class__._bfs_states(
  592|                 self.initial_state, lambda state: iter(self.transitions[state].items())
  593|             )
  594|             reachable_states = set(bfs_states)
  595| 
  596|         reachable_final_states = self.final_states & reachable_states
  597| 
  598|         return self.__class__._minify(
  599|             reachable_states=reachable_states,
  600|             input_symbols=self.input_symbols,
  601|             transitions=self.transitions,
  602|             initial_state=self.initial_state,
  603|             reachable_final_states=reachable_final_states,
  604|             retain_names=retain_names,
  605|         )
